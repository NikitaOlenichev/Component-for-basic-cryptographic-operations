set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(example_hash example_hash.cpp)
add_executable(example_encryption example_encryption.cpp)
add_executable(example_decryption example_decryption.cpp)
add_executable(example_signature example_signature.cpp)

target_include_directories(example_hash PRIVATE ${CMAKE_SOURCE_DIR}/include)
target_include_directories(example_encryption PRIVATE ${CMAKE_SOURCE_DIR}/include)
target_include_directories(example_decryption PRIVATE ${CMAKE_SOURCE_DIR}/include)
target_include_directories(example_signature PRIVATE ${CMAKE_SOURCE_DIR}/include)

target_link_libraries(example_hash PRIVATE Component_for_basic_cryptographic_operations)
target_link_libraries(example_encryption PRIVATE Component_for_basic_cryptographic_operations)
target_link_libraries(example_decryption PRIVATE Component_for_basic_cryptographic_operations)
target_link_libraries(example_signature PRIVATE Component_for_basic_cryptographic_operations)

install(TARGETS example_hash example_encryption example_decryption example_signature
        RUNTIME DESTINATION bin)
